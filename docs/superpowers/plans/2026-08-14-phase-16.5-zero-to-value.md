# Phase 16.5 — Zero-to-Value: Context Proof Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship a `try` subcommand that generates a read-only context selection for any coding objective and prints a human-readable terminal report — no initialization required, no file writes, no provider API or update-check network calls once the package is available locally.

**Architecture:** New module `bin/lib/try.ts` with five exported functions:
- `normalizeCuratedGraph(raw: unknown): CodeGraph` — pure; validates/rebuilds any graph shape field-by-field so all six spread-iterated fields (`tags`, `entrypoints`, `public_contracts`, `dependencies`, `dependents`, `owners`) are always string arrays
- `parseTryObjective(args: string[]): string | null` — pure, extracts objective from rawArgs
- `buildTryReport(...)` — testable with injectable filesystem access
- `formatTryOutput(report: TryReport): string` — pure, sanitizes terminal text for all user-visible strings (objective, path, reason, language fallback)
- `runTry(): void` — impure runner

CLI dispatch gated on `trySubcommand = rawArgs[0] === 'try'`. `skipUpdateCheck` in `context.ts` expanded to include `rawArgs[0] === 'try'`, preventing npm checks, upgrade prompts, or writes during `try`.

**Tech Stack:** Node.js built-in `test`, TypeScript, existing modules — no new npm dependencies. No external color libraries.

## Global Constraints

- No file writes in the `try` path — `generateDependencyGraph` reads files; `runRefreshCodeGraph` must never be called.
- After the package is available locally, `try` makes no network calls — no npm registry check, no provider API calls, no update-check. (The `npx @latest` invocation shown in README may resolve the package from npm, but that is outside the `try` subcommand itself.)
- No new npm dependencies. Use plain Unicode (`─`, `·`) only.
- `npm test` must pass (`npm run typecheck && npm run build && node --import tsx --test test/*.test.ts`).
- User owns all commits — plan includes `git add` commands but no `git commit`.
- Version bump to `3.14.0` using `npm version --no-git-tag-version` (updates both `package.json` and `package-lock.json`).

```typescript
import fs from 'node:fs';
import path from 'node:path';
import type { CodeGraph, DependencyGraph } from './types.js';
import { root, rawArgs } from './context.js';
import {
  selectContextForObjective,
  tryReadCuratedCodeGraph,
  type ContextPackOptions,
  type SelectedContextNode,
} from './context-pack.js';
import { generateDependencyGraph } from './dependency-graph.js';
import { detectProjectProfile, getAvailableProfiles } from './profiles.js';
import { resolveExclusions } from './profile-exclusions.js';
import { getErrorMessage } from './utils.js';

export type TryReport = {
  objective: string;
  terms: string[];
  languages: Map<string, number>;
  totalFiles: number;
  selected: SelectedContextNode[];
  omittedCount: number;
  selectedSourceBytes: number;
};

// ── sanitization ───────────────────────────────────────────────────────────────

/** Strips ANSI escape sequences and control characters from terminal output. */
function sanitizeTerminalText(value: string): string {
  return value
    .replace(/\x1b\[[0-9;]*[A-Za-z]/g, '')  // strip ANSI escape sequences
    .replace(/[\x00-\x1f\x7f]/g, ' ')        // replace control chars with space
    .replace(/\s+/g, ' ')
    .trim();
}

// ── normalizeCuratedGraph ──────────────────────────────────────────────────────

function normalizeString(v: unknown): string | undefined {
  return typeof v === 'string' ? v : undefined;
}

function normalizeStringArray(v: unknown): string[] {
  if (!Array.isArray(v)) return [];
  return v.filter((item): item is string => typeof item === 'string');
}

/**
 * Validates and rebuilds any unknown value into a CodeGraph safe for
 * selectContextForObjective. All six spread-iterated fields (tags, entrypoints,
 * public_contracts, dependencies, dependents, owners) are always string[].
 * Wrong-type scalar fields become undefined.
 */
export function normalizeCuratedGraph(raw: unknown): CodeGraph {
  if (raw === null || raw === undefined || typeof raw !== 'object' || Array.isArray(raw)) {
    return { nodes: [], edges: [] } as unknown as CodeGraph;
  }
  const obj = raw as Record<string, unknown>;

  const nodes = Array.isArray(obj.nodes)
    ? obj.nodes
        .filter((n): n is Record<string, unknown> => n !== null && typeof n === 'object' && !Array.isArray(n))
        .map((n) => ({
          id: normalizeString(n.id),
          path: normalizeString(n.path),
          type: normalizeString(n.type),
          summary: normalizeString(n.summary),
          confidence: normalizeString(n.confidence),
          owners: normalizeStringArray(n.owners),
          entrypoints: normalizeStringArray(n.entrypoints),
          public_contracts: normalizeStringArray(n.public_contracts),
          dependencies: normalizeStringArray(n.dependencies),
          dependents: normalizeStringArray(n.dependents),
          tags: normalizeStringArray(n.tags),
        }))
    : [];

  const edges = Array.isArray(obj.edges)
    ? obj.edges.filter((e): e is Record<string, unknown> => e !== null && typeof e === 'object' && !Array.isArray(e))
    : [];

  return { ...obj, nodes, edges } as unknown as CodeGraph;
}

// ── parseTryObjective ──────────────────────────────────────────────────────────

/** Extracts objective from rawArgs where args[0] === 'try'. */
export function parseTryObjective(args: string[]): string | null {
  const positional = args[1];
  if (positional !== undefined && !positional.startsWith('--')) return positional;
  for (let i = 1; i < args.length; i++) {
    if (args[i] === '--objective') return args[i + 1] ?? null;
    if (args[i].startsWith('--objective=')) return args[i].slice('--objective='.length);
  }
  return null;
}

// ── buildTryReport ─────────────────────────────────────────────────────────────

/**
 * Testable: accepts injectable fileSizer so tests never touch the filesystem.
 * Default fileSizer reads file size from disk relative to root.
 * Reports selectedSourceBytes (raw bytes, not compiled token budget).
 */
export function buildTryReport(
  objective: string,
  depGraph: DependencyGraph,
  curatedGraph: CodeGraph,
  options: ContextPackOptions,
  fileSizer: (relativePath: string) => number = (p) => {
    try { return fs.statSync(path.join(root, p)).size; } catch { return 0; }
  }
): TryReport {
  const safeObjective = sanitizeTerminalText(objective);

  const { terms, selected, omitted } = selectContextForObjective(safeObjective, curatedGraph, depGraph, options);

  const languages = new Map<string, number>();
  for (const node of depGraph.nodes) {
    const lang = node.language ?? 'typescript';
    languages.set(lang, (languages.get(lang) ?? 0) + 1);
  }

  const selectedSourceBytes = selected.reduce((sum, s) => sum + fileSizer(s.node.path), 0);

  return {
    objective: safeObjective,
    terms,
    languages,
    totalFiles: depGraph.nodes.length,
    selected,
    omittedCount: omitted.length,
    selectedSourceBytes,
  };
}

// ── formatTryOutput ────────────────────────────────────────────────────────────

const LANGUAGE_DISPLAY: Record<string, string> = {
  typescript: 'TypeScript',
  javascript: 'JavaScript',
  python: 'Python',
  go: 'Go',
};

function displayLanguage(id: string): string {
  return LANGUAGE_DISPLAY[id] ?? sanitizeTerminalText(id.charAt(0).toUpperCase() + id.slice(1));
}

function formatBytes(n: number): string {
  if (n >= 1024 * 1024) return `${(n / (1024 * 1024)).toFixed(1)} MB`;
  if (n >= 1024) return `${(n / 1024).toFixed(1)} KB`;
  return `${n} bytes`;
}

/** Pure: formats the report as a terminal string. Sanitizes objective, path, reason, and language display. */
export function formatTryOutput(report: TryReport): string {
  const lines: string[] = [];

  // Sanitize objective defensively even though buildTryReport already does it,
  // so formatTryOutput upholds its own contract when called directly with a
  // manually constructed TryReport.
  const safeObjective = sanitizeTerminalText(report.objective);
  lines.push(`ForgeAI context proof — "${safeObjective}"`);
  lines.push('');

  const langEntries = [...report.languages.entries()].sort((a, b) => b[1] - a[1] || a[0].localeCompare(b[0]));
  const langDisplay = langEntries.map(([id]) => displayLanguage(id)).join(' · ');
  const fileWord = report.totalFiles === 1 ? 'source file' : 'source files';
  lines.push(`  Detected    ${langDisplay || '(none)'}  (${report.totalFiles} ${fileWord})`);

  const selectedCount = report.selected.length;
  lines.push(
    `  Relevant    ${
      selectedCount === 0
        ? 'none — no objective-matched files'
        : `${selectedCount} file${selectedCount === 1 ? '' : 's'} matched`
    }`
  );

  if (report.omittedCount > 0) {
    lines.push(`  Excluded    ${report.omittedCount} by profile exclusion`);
  }

  if (report.selectedSourceBytes > 0) {
    lines.push(
      `  Source size ~${formatBytes(report.selectedSourceBytes)} selected  (raw source — not compiled excerpts)`
    );
  }

  lines.push('');
  lines.push('  Included files:');
  lines.push('  ' + '─'.repeat(72));

  if (report.selected.length === 0) {
    lines.push('    (no objective-matched files — refine the objective or check source file names)');
  } else {
    const safePaths = report.selected.map((s) => sanitizeTerminalText(s.node.path));
    const maxPathLen = Math.max(...safePaths.map((p) => p.length));
    for (let i = 0; i < report.selected.length; i++) {
      const padded = safePaths[i].padEnd(maxPathLen + 2);
      const safeReason = sanitizeTerminalText(report.selected[i].reason);
      const reason = safeReason.length > 80 ? safeReason.slice(0, 77) + '...' : safeReason;
      lines.push(`    ${padded}${reason}`);
    }
  }

  // CTA: three-step install flow. Objective is NOT embedded in any shell command
  // to prevent injection when users copy-paste.
  lines.push('');
  lines.push('  Ready to use ForgeAI on this repository?');
  lines.push('    npx forgeai-agentic-init@latest --profile auto');
  lines.push('    npx forgeai-agentic-init@latest --refresh-codegraph');
  lines.push('    npx forgeai-agentic-init@latest --compile-context --objective "<your objective>"');
  lines.push('');

  return lines.join('\n');
}

// ── runTry ─────────────────────────────────────────────────────────────────────

export function runTry(): void {
  const objective = parseTryObjective(rawArgs);
  if (!objective || objective.trim() === '') {
    process.stderr.write(
      'Usage: forgeai-init try "<objective>"\n' +
      '       forgeai-init try --objective "<description>"\n'
    );
    process.exitCode = 2;
    return;
  }

  let depGraph: DependencyGraph;
  try {
    depGraph = generateDependencyGraph(root);
  } catch (error) {
    process.stderr.write(`Error: could not analyze source files (${getErrorMessage(error)}).\n`);
    process.exitCode = 1;
    return;
  }

  // Normalize curated graph field-by-field; falls back to empty on any malformed shape.
  const curatedGraph = normalizeCuratedGraph(tryReadCuratedCodeGraph(root));

  // Auto-detect primary project profile — matches --profile auto behavior.
  // Applies matching exclusions so preview reflects post-install context-pack results.
  const primaryProfile = detectProjectProfile();
  const availableSet = new Set(getAvailableProfiles());
  const components = primaryProfile !== null && availableSet.has(primaryProfile) ? [primaryProfile] : [];
  const rules = resolveExclusions(components);

  const report = buildTryReport(objective, depGraph, curatedGraph, {
    maxNodes: 12,
    maxDepth: 2,
    rules,
    includeGlobs: [],
  });
  process.stdout.write(formatTryOutput(report));
}
```

```typescript
// runDist throws on non-zero exit. This variant captures exit code and output.
function runDistResult(
  args: string[],
  cwd: string,
  extraEnv: Record<string, string> = {}
): { stdout: string; stderr: string; status: number } {
  try {
    const stdout = execFileSync(process.execPath, [distCli, ...args], {
      cwd,
      encoding: 'utf8',
      env: { ...process.env, FORGEAI_SKIP_UPDATE_CHECK: '1', ...extraEnv },
    }) as string;
    return { stdout, stderr: '', status: 0 };
  } catch (err: unknown) {
    const e = err as { stdout?: string; stderr?: string; status?: number };
    return { stdout: e.stdout ?? '', stderr: e.stderr ?? '', status: e.status ?? 1 };
  }
}

/**
 * Snapshots relative-path → SHA-256(content) for every file in dir.
 * Catches create, delete, and overwrite — not just path changes.
 */
function snapshotFiles(dir: string): Map<string, string> {
  const result = new Map<string, string>();
  function walk(d: string) {
    for (const entry of fs.readdirSync(d, { withFileTypes: true })) {
      const full = path.join(d, entry.name);
      if (entry.isDirectory()) {
        walk(full);
      } else {
        const rel = path.relative(dir, full);
        const hash = crypto.createHash('sha256').update(fs.readFileSync(full)).digest('hex');
        result.set(rel, hash);
      }
    }
  }
  walk(dir);
  return result;
}
```

## Performance Note (P3)

`generateDependencyGraph` parses all source files synchronously. On very large repos, this may be slow. This is a **known limitation** for this MVP: there is no per-command timing measurement and no acceptance criterion bounding `try` execution time independently of the full test suite. A formal benchmark (e.g., "1 000 source files < 10 s on CI") is deferred to a follow-up phase if activation data shows repos of that size are a common entry point for `try`.
